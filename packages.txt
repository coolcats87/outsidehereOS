# Packages install.sh installs. One per line; "#" starts a comment.
# install.sh works out which come from the Arch repos and which from the AUR.
# Already-installed packages (or ones something else already provides) are skipped.

hyprland                      # the compositor (Lua config, 0.57+)

hyprpicker                    # colour picker (Super+Shift+C)
hyprsunset                    # night light
xdg-desktop-portal-hyprland   # screen sharing, file pickers
xdg-desktop-portal-gtk

qt6-wayland
qt6-svg
qt6-imageformats              # webp/avif wallpapers and covers
qt6-multimedia
qt6-5compat

ydotool                       # pastes for you
jq                            # reads/writes the settings JSON
libqalculate                  # math in Spotlight (qalc)
